STEP 1: GET GOOGLE CREDENTIALS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Go to https://console.cloud.google.com/
   2. Create new project → Name: "MonOrienta"
   3. Enable Google+ API in Library
   4. Create OAuth 2.0 Client ID credentials
   5. Add redirect URLs:
      ├─ http://localhost:3000/auth/callback
      └─ https://your-project.vercel.app/auth/callback
   6. Copy Client ID ⬜ and Client Secret ⬜


STEP 2: ENABLE IN SUPABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Go to https://app.supabase.com/
   2. Select MonOrienta project
   3. Authentication → Providers → Google
   4. Toggle Enable ✅
   5. Paste:
      ├─ Client ID ⬜
      └─ Client Secret ⬜
   6. Copy Supabase redirect URL
   7. Add to Google OAuth redirect URLs
   8. Save ✅


STEP 3: TEST LOCALLY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SESSION PERSISTENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   User Logs In
        ↓
   Session stored in HTTP-Only Cookie
        ↓
   User closes browser
        ↓
   Cookie persists
        ↓
   User reopens app
        ↓
   Middleware checks cookie
        ↓
   Session valid?
        ├─ YES → User automatically logged in
        └─ NO → Redirect to login


CROSS-DEVICE LOGIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Device A                          Device B
   │                                  │
   User Signs In                      │
   │ clicks Google                    │
   │  ↓                               │
   ├→ Authenticated                   │
   │  Session stored                  │
   │                                  User Opens App
   │                                  │ clicks Google
   │                                  │  ↓
   │                                  ├→ Google recognizes user
   │                                  │  New session for Device B
   │                                  │  ↓
   ├→ Device A Logged In              ├→ Device B Logged In
   │  Independent session             │  Independent session
   │                                  │
   User logout on A                   │
   │                                  │
   ├→ Device A session cleared        │
   │  Device A: Logged Out            │
   │                                  │
   │                                  Device B still logged in!
   │                                  Device B session unaffected
